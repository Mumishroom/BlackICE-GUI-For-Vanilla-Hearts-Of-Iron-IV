# Special forces in use / limit, vanilla defines (SPECIAL_FORCES_CAP_BASE 0.05, SPECIAL_FORCES_CAP_MIN 24)
ff_calc_special_forces = {
	# raw number of special forces battalions
	set_temp_variable = { ff_sf_raw = num_battalions_with_type@paratrooper }
	add_to_temp_variable = { ff_sf_raw = num_battalions_with_type@marine }
	add_to_temp_variable = { ff_sf_raw = num_battalions_with_type@mountaineers }
	# weighted by *_special_forces_contribution_factor
	set_temp_variable = { ff_sf_used = 0 }
	set_temp_variable = { ff_sf_k = modifier@paratroopers_special_forces_contribution_factor }
	add_to_temp_variable = { ff_sf_k = 1 }
	set_temp_variable = { ff_sf_w = num_battalions_with_type@paratrooper }
	multiply_temp_variable = { ff_sf_w = ff_sf_k }
	add_to_temp_variable = { ff_sf_used = ff_sf_w }
	set_temp_variable = { ff_sf_k = modifier@marines_special_forces_contribution_factor }
	add_to_temp_variable = { ff_sf_k = 1 }
	set_temp_variable = { ff_sf_w = num_battalions_with_type@marine }
	multiply_temp_variable = { ff_sf_w = ff_sf_k }
	add_to_temp_variable = { ff_sf_used = ff_sf_w }
	set_temp_variable = { ff_sf_k = modifier@mountaineers_special_forces_contribution_factor }
	add_to_temp_variable = { ff_sf_k = 1 }
	set_temp_variable = { ff_sf_w = num_battalions_with_type@mountaineers }
	multiply_temp_variable = { ff_sf_w = ff_sf_k }
	add_to_temp_variable = { ff_sf_used = ff_sf_w }
	round_temp_variable = ff_sf_used
	# limit = max(MIN, BASE * regular battalions) + flat
	set_temp_variable = { ff_sf_cap = num_battalions }
	subtract_from_temp_variable = { ff_sf_cap = ff_sf_raw }
	set_temp_variable = { ff_sf_ratio = 0.05 }
	add_to_temp_variable = { ff_sf_ratio = modifier@special_forces_cap }
	multiply_temp_variable = { ff_sf_cap = ff_sf_ratio }
	subtract_from_temp_variable = { ff_sf_cap = 0.499 }
	round_temp_variable = ff_sf_cap
	set_temp_variable = { ff_sf_min = 24 }
	add_to_temp_variable = { ff_sf_min = modifier@special_forces_min }
	clamp_temp_variable = { var = ff_sf_cap min = ff_sf_min max = 999999 }
	add_to_temp_variable = { ff_sf_cap = modifier@special_forces_cap_flat }
	always = yes
}
