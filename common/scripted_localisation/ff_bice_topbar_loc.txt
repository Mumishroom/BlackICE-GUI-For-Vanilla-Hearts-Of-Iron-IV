# Faction Freedom: BlackICE-style top bar number formatting
# (logic adapted from BlackICE BI scripted localisation, uses only vanilla variables)

defined_text = {
	name = FF_GetTopbarPolitical
	text = {
		trigger = { has_political_power > 99.9 }
		localization_key = FF_TOPBAR_PP_0
	}
	text = {
		trigger = { has_political_power > 9.9 }
		localization_key = FF_TOPBAR_PP_1
	}
	text = {
		localization_key = FF_TOPBAR_PP_2
	}
}

defined_text = {
	name = FF_GetTopbarManpower
	text = {
		trigger = {
			check_variable = { manpower_k > 10000 }
			set_temp_variable = { ff_mp_m = manpower_k }
			divide_temp_variable = { ff_mp_m = 1000 }
		}
		localization_key = FF_TOPBAR_MP_M2
	}
	text = {
		trigger = {
			check_variable = { manpower_k > 1000 }
			set_temp_variable = { ff_mp_m = manpower_k }
			divide_temp_variable = { ff_mp_m = 1000 }
		}
		localization_key = FF_TOPBAR_MP_M3
	}
	text = {
		trigger = { check_variable = { manpower_k > 10 } }
		localization_key = FF_TOPBAR_MP_K
	}
	text = {
		localization_key = FF_TOPBAR_MP
	}
}
